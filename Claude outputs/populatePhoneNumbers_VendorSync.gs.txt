  function normZip_(z) {
    return String(z || '').replace(/[^0-9]/g, '').slice(0, 5);
  }

  function streetNum_(a) {
    var m = String(a || '').match(/^\s*(\d+)/);
    return m ? m[1] : '';
  }

  var STOP = {
    suite: 1, ste: 1, unit: 1, bldg: 1, building: 1, the: 1, of: 1, and: 1,
    dr: 1, drive: 1, rd: 1, road: 1, st: 1, street: 1, blvd: 1, ln: 1,
    pkwy: 1, parkway: 1, ave: 1, avenue: 1
  };

  function normWords_(s) {
    return String(s || '')
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, ' ')
      .split(' ')
      .filter(function (w) { return w.length > 1 && !STOP[w]; });
  }

  function jaccard_(aWords, bWords) {
    var aSet = {}, bSet = {}, union = {}, inter = 0;
    aWords.forEach(function (w) { aSet[w] = true; });
    bWords.forEach(function (w) { bSet[w] = true; });
    Object.keys(aSet).forEach(function (w) { union[w] = true; if (bSet[w]) inter++; });
    Object.keys(bSet).forEach(function (w) { union[w] = true; });
    var size = Object.keys(union).length;
    return size ? inter / size : 0;
  }
